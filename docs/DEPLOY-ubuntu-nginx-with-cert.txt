================================================================================
 IT SERVICE DESK - DEPLOY ขึ้น UBUNTU + NGINX (แบบ HTTPS / ใช้ Cert)
================================================================================
 Stack: Laravel 12 + React/Vite + Sanctum cookie SPA + MySQL
        Database driver (session/cache/queue) + Scheduler 2 งาน + DomPDF
 Cert  : Let's Encrypt (ฟรี, auto-renew) ผ่าน certbot
 สมมติ : Ubuntu 22.04/24.04, deploy ที่ /var/www/itservices
 *** ข้อกำหนด: ต้องมี "โดเมนจริง" ที่ชี้ A record มาที่ IP ของ server แล้ว ***
         (Let's Encrypt ออก cert ให้ IP ล้วนไม่ได้ ต้องเป็นโดเมน เช่น itsd.inaba.co.th)
================================================================================

 4. ตั้งค่า .env  (ค่าสำหรับ HTTPS)
--------------------------------------------------------------------------------

cp .env.example .env
nano .env

  # ----- แก้ค่าสำคัญ -----
  APP_NAME="IT Service Desk"
  APP_ENV=production
  APP_KEY=                                   # << ปล่อยว่างไว้ก่อน, step 5 รัน key:generate ให้
  APP_DEBUG=false                            # << production ต้อง false (โชว์หน้า error สวย ไม่โชว์ stack trace)
  APP_URL=https://itsd.example.com           # << ใส่ "โดเมนจริง" + https

  APP_LOCALE=th
  APP_TIMEZONE=UTC

  LOG_LEVEL=error                            # << production ใช้ error (อย่าใช้ debug = เปลือง disk + เสี่ยงข้อมูลหลุด)

  # --- Database (MySQL) ---
  # *** ระวัง: ต้องเป็น DB_CONNECTION (ไม่ใช่ B_CONNECTION) ***
  #     ถ้าพิมพ์ผิด Laravel จะ fallback ไป sqlite -> ต่อ MySQL ไม่ติด แอปล่ม
  DB_CONNECTION=mysql
  DB_HOST=127.0.0.1
  DB_PORT=3306
  DB_DATABASE=itservices
  DB_USERNAME=itservices
  DB_PASSWORD=รหัสผ่านที่ตั้งไว้             # << ตรงกับที่ตั้งใน step 2 (อย่า commit รหัสจริงขึ้น git)

  # --- Session / Sanctum SPA บน HTTPS ---
  SESSION_DRIVER=database
  SESSION_DOMAIN=itsd.example.com           # โดเมนเดียวกับ APP_URL (ไม่ต้องมี https://)
  SESSION_SECURE_COOKIE=true                # << true เพราะมี HTTPS แล้ว (ปลอดภัยกว่า)
  SANCTUM_STATEFUL_DOMAINS=itsd.example.com

  QUEUE_CONNECTION=database
  CACHE_STORE=database

  # Mail: โปรเจกต์อ่านค่าจาก table mail_settings เป็นหลัก
  # .env เป็นแค่ fallback -- ตั้งค่าจริงผ่านหน้า Settings
  MAIL_MAILER=log

  # --- บัญชีผู้ดูแลที่ db:seed จะสร้าง (step 5) ---
  # ถ้าไม่ตั้ง 3 ตัวนี้: ชื่อจะเป็น "Administrator", ไม่มีอีเมล, และรหัสจะถูกสุ่ม
  # แล้วพิมพ์ออก console ครั้งเดียว (ยังใช้ได้ แต่ต้องคัดลอกไว้ทันที)
  # *** ต้องตั้งก่อนรัน db:seed และก่อน optimize ***  ดู step 5
  SEED_SUPER_NAME=super
  SEED_SUPER_EMAIL=admin@example.com         # << อีเมลจริง -- การแจ้งเตือนส่งเฉพาะบัญชีที่มีอีเมล
  SEED_SUPER_PASSWORD=                       # << ตั้งรหัสชั่วคราว (ระบบบังคับเปลี่ยนตอน login ครั้งแรก)

# *** จำเป็น ไม่ใช่ทางเลือก *** migrate สร้างแต่ตาราง ตาราง users จะว่าง
# ทำให้ login ไม่ได้เลย db:seed คือตัวที่สร้าง:
#   - บัญชี super (บัญชีเดียวที่มี -- ไม่มีตัวนี้ก็เข้าระบบไม่ได้)
#   - roles + role_permissions        (ไม่มี = กำหนดสิทธิ์ไม่ได้)
#   - email templates 26 ตัว          (ไม่มี = เมลไม่ออกแม้ตั้ง SMTP ถูก)
#   - workflows + steps               (ไม่มี = ส่งคำขอไม่ได้)
#   - request_options                 (ไม่มี = คำขอ 3 ประเภทมีฟิลด์ required ที่เลือกไม่ได้)
# ไม่ใส่ --class => เรียก DatabaseSeeder (production) เท่านั้น
# *** ห้ามใช้ --class=DemoSeeder บน production *** (สร้างบัญชีรหัส "password" 5 บัญชี)

                                  # เพราะพอ cache config แล้ว SEED_SUPER_* จะอ่านไม่เจอ
                                  # (db:seed จะสุ่มรหัสให้แทนค่าที่ตั้งใน .env)

--------------------------------------------------------------------------------
 7. ออก CERT ด้วย cloud flare 
--------------------------------------------------------------------------------

1. Login dash.cloudflare.com
2. เลือก Domain → SSL/TLS → Origin Server
3. คลิก Create Certificate
4. ตั้งค่า:
- Key type: RSA (2048)
- Hostnames: yourdomain.com และ *.yourdomain.com
- Expiration: 15 years (แนะนำ)
5. คลิก Create
6. Copy ทั้ง 2 อัน (จะดูได้ครั้งเดียว!):
- Origin Certificate → cert.pem
- Private Key → key.pem

 8. ตั้งค่า NGINX (เริ่มจาก HTTP ก่อน เดี๋ยว certbot จะเติม HTTPS ให้)
--------------------------------------------------------------------------------

sudo nano /etc/nginx/sites-available/itservices


server {
      listen 80;
      server_name it.ink-connect.com;            # << โดเมนจริง
      return 301 https://$host$request_uri;
}

server {
      listen 443 ssl;
      server_name it.ink-connect.com;

      root /var/www/itservices/public;
      index index.html index.php;

      # Cloudflare Origin Certificate
      ssl_certificate     /etc/ssl/cloudflare/cert.pem;
      ssl_certificate_key /etc/ssl/cloudflare/key.pem;

      ssl_protocols       TLSv1.2 TLSv1.3;
      ssl_ciphers         HIGH:!aNULL:!MD5;
      ssl_prefer_server_ciphers on;

      charset utf-8;
      client_max_body_size 50M;

      # Laravel

      location / {
          try_files $uri $uri/ /index.php?$query_string;
      }

      location ~ \.php$ {
          fastcgi_pass unix:/var/run/php/php8.3-fpm.sock;
          fastcgi_index index.php;
          fastcgi_param SCRIPT_FILENAME $realpath_root$fastcgi_script_name;
          include fastcgi_params;
      }

      location ~ /\.(?!well-known).* {
          deny all;
      }

      location = /favicon.ico { access_log off; log_not_found off; }
      location = /robots.txt  { access_log off; log_not_found off; }

      # Cloudflare Real IP
      set_real_ip_from 103.21.244.0/22;
      set_real_ip_from 103.22.200.0/22;
      set_real_ip_from 103.31.4.0/22;
      set_real_ip_from 104.16.0.0/13;
      set_real_ip_from 104.24.0.0/14;
      set_real_ip_from 108.162.192.0/18;
      set_real_ip_from 131.0.72.0/22;
      set_real_ip_from 141.101.64.0/18;
      set_real_ip_from 162.158.0.0/15;
      set_real_ip_from 172.64.0.0/13;
      set_real_ip_from 173.245.48.0/20;
      set_real_ip_from 188.114.96.0/20;
      set_real_ip_from 190.93.240.0/20;
      set_real_ip_from 197.234.240.0/22;
      set_real_ip_from 198.41.128.0/17;
      real_ip_header CF-Connecting-IP;
}



# เปิดใช้งาน + reload

--------------------------------------------------------------------------------
 9. QUEUE WORKER (ส่งอีเมล/แจ้งเตือนผ่าน queue)
--------------------------------------------------------------------------------

sudo nano /etc/systemd/system/itservices-queue.service

  # ----- เนื้อหาไฟล์ -----
  [Unit]
  Description=IT Service Desk Queue Worker
  After=network.target mysql.service

  [Service]
  User=www-data
  Group=www-data
  Restart=always
  ExecStart=/usr/bin/php /var/www/itservices/artisan queue:work --sleep=3 --tries=3 --max-time=3600
  WorkingDirectory=/var/www/itservices

  [Install]
  WantedBy=multi-user.target

sudo systemctl daemon-reload
sudo systemctl enable --now itservices-queue
sudo systemctl status itservices-queue

================================================================================
 ข้อควรระวัง (deploy แบบมี cert)
================================================================================

 1. ต้องมีโดเมนจริงชี้ A record มาที่ IP ของ server "ก่อน" รัน certbot
    ไม่งั้น Let's Encrypt verify ไม่ผ่าน (ใช้ IP ล้วนไม่ได้)

 2. เปิด port 80 และ 443 ที่ firewall / cloud security group:
      sudo ufw allow 'Nginx Full'     (ถ้าใช้ ufw)

 3. SESSION_SECURE_COOKIE=true บังคับให้ cookie ส่งผ่าน HTTPS เท่านั้น
    -> ถ้ายังเข้าผ่าน http:// อยู่ จะ login ไม่ติด ต้องเข้าผ่าน https:// เสมอ

 4. APP_URL และ SANCTUM_STATEFUL_DOMAINS ต้องเป็นโดเมน "ตัวเดียวกัน" กับที่ออก cert
    ถ้าไม่ตรง -> CORS / cookie ใช้ไม่ได้

 5. cert ต่ออายุอัตโนมัติทุก ~60 วัน (อายุจริง 90 วัน) ผ่าน certbot timer
    ตรวจสุขภาพด้วย: sudo certbot renew --dry-run
